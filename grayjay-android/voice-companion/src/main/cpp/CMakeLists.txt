cmake_minimum_required(VERSION 3.22)
project(grayjoy_voice LANGUAGES C CXX)
include(FetchContent)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "x86_64")
    # A portable APK must not assume AVX support on older Android x86-64 devices.
    set(GGML_AVX OFF CACHE BOOL "" FORCE)
    set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
    set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
    set(GGML_FMA OFF CACHE BOOL "" FORCE)
    set(GGML_F16C OFF CACHE BOOL "" FORCE)
endif()
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(llama
    # Pinned archive keeps native builds reproducible across local and CI environments.
    URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/8bb3cf0d216a5ae48360179f2931daf284be4d01
    URL_HASH SHA256=1b1a5bd079840425683ff6caebf3543e08d2dbcab7563782fabdf145030521a2)
# Expose llama's target and transitive headers to the JNI bridge.
FetchContent_MakeAvailable(llama)
add_library(grayjoy-voice SHARED voice_llm.cpp)
target_compile_options(grayjoy-voice PRIVATE -O3)
target_link_libraries(grayjoy-voice PRIVATE llama android log)
